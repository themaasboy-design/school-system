    <script>
        const container = document.getElementById('classesContainer');
        for(let i=1; i<=7; i++) {
            container.innerHTML += `
                <div class="class-box">
                    <label>الحصة ${i}:</label>
                    <select name="class${i}_room" id="roomSelect_${i}" onchange="checkAndEnableTeacher(${i})">
                        <option value="">-- اختر الفصل --</option>
                    </select>
                    <select name="class${i}_teacher" id="teacherSelect_${i}" disabled>
                        <option value="">-- اختر الفصل أولاً --</option>
                    </select>
                </div>
            `;
        }

        document.getElementById('absenceForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const reportData = {
                date: document.getElementById('hijriDate').value,
                day: document.getElementById('daySelect').value,
                absentTeacher: document.getElementById('teacherSelect').value
            };
            for(let i=1; i<=7; i++) {
                reportData[`class${i}_room`] = document.getElementById(`roomSelect_${i}`).value;
                reportData[`class${i}_teacher`] = document.getElementById(`teacherSelect_${i}`).value;
            }
            try {
                const response = await fetch('/save-report', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify(reportData)
                });
                const result = await response.json();
                if (result.success) {
                    alert("تم حفظ بيانات الغياب والانتظار بنجاح في ملف الإكسيل!");
                } else {
                    alert("خطأ: " + result.error);
                }
            } catch (err) { alert("فشل الاتصال بالسيرفر"); }
        });

        async function loadClasses() {
            try {
                const response = await fetch('/get-classes');
                const classes = await response.json();
                for(let i=1; i<=7; i++) {
                    let roomSelect = document.getElementById(`roomSelect_${i}`);
                    classes.forEach(row => {
                        let opt = document.createElement('option');
                        opt.value = row["Name_class"];
                        opt.text = row["Name_class"];
                        roomSelect.appendChild(opt);
                    });
                }
            } catch (e) { console.error("خطأ في جلب الفصول"); }
        }

        async function loadWaitingTeachers(i) {
            const day = document.getElementById('daySelect').value;
            const roomSelect = document.getElementById(`roomSelect_${i}`);
            const teacherSelect = document.getElementById(`teacherSelect_${i}`);

            if (!day) { alert("يرجى اختيار اليوم أولاً"); roomSelect.value = ""; return; }
            if (roomSelect.value === "") { teacherSelect.disabled = true; return; }

            try {
                const response = await fetch(`/get-waiting-teachers?day=${day}&period=${i}`);
                const data = await response.json();
                teacherSelect.innerHTML = '<option value="">-- اختر المعلم البديل --</option>';
                if (data.teachers && data.teachers.length > 0) {
                    data.teachers.forEach(teacherCode => {
                        let opt = document.createElement('option');
                        opt.value = teacherCode;
                        opt.text = teacherCode; 
                        teacherSelect.appendChild(opt);
                    });
                    teacherSelect.disabled = false;
                } else {
                    teacherSelect.innerHTML = '<option value="">-- لا يوجد معلمين متاحين --</option>';
                    teacherSelect.disabled = true;
                }
            } catch (e) { console.error("خطأ في جلب المعلمين البدلاء"); }
        }

        function checkAndEnableTeacher(i) { loadWaitingTeachers(i); }

        async function loadAllTeachers() {
            try {
                const response = await fetch('/get-data');
                const teachers = await response.json();
                const mainSelect = document.getElementById('teacherSelect');
                teachers.forEach(t => {
                    let opt = document.createElement('option');
                    opt.value = t["Name_teacher"];
                    opt.text = t["Name_teacher"];
                    mainSelect.appendChild(opt);
                });
            } catch (e) { console.error("خطأ في جلب المعلمين"); }
        }
async function loadSchoolName() {
    try {
        const response = await fetch('/get-report-data');
        const data = await response.json();
        if (data.mainInfo && data.mainInfo.B2) {
            document.getElementById('schoolName').textContent = "مدرسة " + data.mainInfo.B2;
        }
    } catch (e) {
        console.error("خطأ في جلب اسم المدرسة:", e);
    }
}

document.addEventListener("DOMContentLoaded", () => {
    loadSchoolName(); // تفعيل جلب اسم المدرسة
    loadAllTeachers();
    loadClasses();
});
