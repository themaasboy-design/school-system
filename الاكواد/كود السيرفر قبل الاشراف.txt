const express = require('express');
const xlsx = require('xlsx');
const bodyParser = require('body-parser');
const app = express();
const PORT = 3000;

app.use(express.static(__dirname));
app.use(bodyParser.json());

// =========================================================================
// ⚙️ إعدادات أسماء الملف والورقات (تعديلها هنا فقط وسيطبق على كامل المشروع)
// =========================================================================
const EXCEL_FILE = 'waiting_data.xlsx'; // 👈 ضَع اسم ملف الإكسل الجديد هنا

// أوراق العمل (يمكنك كتابة اسم الورقة كـ نص "اسم_الورقة" أو ترك رقم الترتيب)
const TEACHERS_SHEET = 'teacher_name';       // ورقة المعلمين التي تُعرض في Dashboard (الورقة الأولى = 0)
const CLASSES_SHEET ='classes';        // ورقة الفصول (الورقة الثانية = 1)
const WAITING_SHEET = ' Waiting_table';        // ورقة جدول الانتظار (الورقة الثالثة = 2)
const REPORT_SHEET = 'report';   // ورقة حفظ التقارير
const MAIN_INFO_SHEET = 'maininfo'; // ورقة البيانات الأساسية
// =========================================================================

// دالة مساعدة للحصول على الورقة سواءً باسمها أو بترتيبها
function getSheet(workbook, sheetIdentifier) {
    if (typeof sheetIdentifier === 'number') {
        return workbook.Sheets[workbook.SheetNames[sheetIdentifier]];
    }
    return workbook.Sheets[sheetIdentifier];
}

// 1️⃣ مسار جلب المعلمين لصفحة الـ Dashboard
app.get('/get-data', (req, res) => {
    try {
        const workbook = xlsx.readFile(EXCEL_FILE);
        const sheet = getSheet(workbook, TEACHERS_SHEET);
        res.json(xlsx.utils.sheet_to_json(sheet));
    } catch (e) { 
        console.error("خطأ في جلب بيانات المعلمين:", e.message);
        res.status(500).json([]); 
    }
});

// 2️⃣ مسار جلب الفصول
app.get('/get-classes', (req, res) => {
    try {
        const workbook = xlsx.readFile(EXCEL_FILE);
        const sheet = getSheet(workbook, CLASSES_SHEET);
        res.json(xlsx.utils.sheet_to_json(sheet));
    } catch (e) { 
        res.status(500).json([]); 
    }
});

// 3️⃣ مسار جلب المعلمين الأربعة للانتظار بناءً على اليوم وعمود الحصة
app.get('/get-waiting-teachers', (req, res) => {
    const { day, period } = req.query; 
    try {
        const workbook = xlsx.readFile(EXCEL_FILE);
        const sheet = getSheet(workbook, WAITING_SHEET); 
        const data = xlsx.utils.sheet_to_json(sheet, { header: 1 });

        let dayIndex = data.findIndex(row => row && row[0] === day);
        if (dayIndex === -1) return res.json({ teachers: [] });

        const colIndex = parseInt(period) - 1; 
        let resultTeachers = [];
        
        for (let i = 1; i <= 4; i++) {
            let row = data[dayIndex + i];
            if (row && row[colIndex] !== undefined) {
                resultTeachers.push(row[colIndex]);
            }
        }
        res.json({ teachers: resultTeachers });
    } catch (e) { 
        res.status(500).json({ error: "خطأ في قراءة الملف: " + e.message }); 
    }
});

// 4️⃣ مسار حفظ التقرير
app.post('/save-report', (req, res) => {
    const reportData = req.body;
    try {
        const workbook = xlsx.readFile(EXCEL_FILE);
        const header = ["التاريخ", "اليوم", "المعلم الغائب", "فصل 1", "بديل 1", "فصل 2", "بديل 2", "فصل 3", "بديل 3", "فصل 4", "بديل 4", "فصل 5", "بديل 5", "فصل 6", "بديل 6", "فصل 7", "بديل 7"];
        
        let data = [];
        let worksheet;

        if (workbook.SheetNames.includes(REPORT_SHEET)) {
            worksheet = workbook.Sheets[REPORT_SHEET];
            data = xlsx.utils.sheet_to_json(worksheet, { header: 1 });
        } else {
            data.push(header);
        }

        let row = [reportData.date, reportData.day, reportData.absentTeacher];
        for(let i = 1; i <= 7; i++) {
            row.push(reportData[`class${i}_room`], reportData[`class${i}_teacher`]);
        }
        data.push(row);

        const newSheet = xlsx.utils.aoa_to_sheet(data);
        
        if (!workbook.SheetNames.includes(REPORT_SHEET)) {
            xlsx.utils.book_append_sheet(workbook, newSheet, REPORT_SHEET);
        } else {
            workbook.Sheets[REPORT_SHEET] = newSheet;
        }

        xlsx.writeFile(workbook, EXCEL_FILE);
        res.json({ success: true });
    } catch (e) { 
        res.status(500).json({ error: "خطأ أثناء الحفظ: " + e.message }); 
    }
});

// 5️⃣ مسار جلب بيانات التقارير والبيانات الأساسية
app.get('/get-report-data', (req, res) => {
    try {
        const workbook = xlsx.readFile(EXCEL_FILE);
        
        let reportData = [];
        if (workbook.SheetNames.includes(REPORT_SHEET)) {
            reportData = xlsx.utils.sheet_to_json(workbook.Sheets[REPORT_SHEET]);
        }

        let mainInfo = { B1: "", B2: "", B3: "", B4: "" };
        if (workbook.SheetNames.includes(MAIN_INFO_SHEET)) {
            const infoSheet = workbook.Sheets[MAIN_INFO_SHEET];
            const infoData = xlsx.utils.sheet_to_json(infoSheet, { header: 1 });
            mainInfo = {
                B1: infoData[0] ? infoData[0][1] : "", 
                B2: infoData[1] ? infoData[1][1] : "", 
                B3: infoData[2] ? infoData[2][1] : "", 
                B4: infoData[3] ? infoData[3][1] : ""
            };
        }
        
        res.json({ teachers: reportData, mainInfo: mainInfo });
    } catch (e) {
        res.status(500).json({ error: e.message });
    }
});
